rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Who may write: the "boat systems" login (the Pi and both Cloudflare
    // workers) and the app's shared login. Both made in Firebase ->
    // Authentication -> Users; the UIDs are copied from there:
    // first the boat systems login, then the app's shared login.
    function crew() {
      return request.auth != null
          && request.auth.uid in ['hhEF6TtgEjVs2aGEuPuASJeQX513', '4IeeyMKCIoOGqbgvtp3GynSg7BI2'];
    }
    // The public Wix camera widget may ask for a photo or live view: it may
    // change only these three fields, and only to whole numbers (times).
    function cameraRequestOnly() {
      return request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(['requestAt', 'liveRequestAt', 'liveWatchAt'])
          && request.resource.data.get('requestAt', 0) is int
          && request.resource.data.get('liveRequestAt', 0) is int
          && request.resource.data.get('liveWatchAt', 0) is int;
    }

    // Anyone may READ everything - the boat's position is public on purpose,
    // and the Wix widgets and the cabin display only read.
    // Only crew may WRITE: the Pi, the Cloudflare workers, a signed-in app.
    match /{collection}/{doc} {
      allow read: if true;
      allow write: if crew();
    }

    // The one public write: the Wix widget (or a signed-out app) asking the
    // deck camera for a photo or live view.
    match /camera/harvest-moon {
      allow update: if cameraRequestOnly();
    }
  }
}
